\begin{block}{\Large Motivation: Inference-Time Compute for RL}
\textbf{Lesson from LLMs:} thinking models scale compute \emph{at inference time}.
Standard RL amortizes all decision-making into one policy forward pass.

\vspace{0.4em}
\textbf{Diffusion policies} model $\pi_\theta(a|s)$ as iterative denoising:
\[
a_T \sim \mathcal{N}(0, I) \;\to\; a_{T-1} \;\to\; \cdots \;\to\; a_0 \quad (\text{conditioned on } s)
\]
Each reverse step admits a gradient: we can \textbf{steer the chain at inference time}.

\vspace{0.4em}
\textbf{Goal.} A self-improvement loop driven by inference-time compute:
\begin{itemize}
  \item \textcolor{harvardcrimson}{\textbf{Sample}} $a\sim\pi'(a|s)\propto\pi_\theta(a|s)\,e^{\eta A(s,a)}$ via MALA-guided diffusion.
  \item \textcolor{softgreen}{\textbf{Distill}} the tilted samples back into $\pi_\theta$ by behavior cloning.
  \item \textcolor{harvardcrimson}{\textbf{Repeat}} -- $Q$ tracks the moving policy via TD learning.
\end{itemize}
\end{block}

\vspace{0.6cm}

\begin{block}{\Large The Target: A Well-Defined Tilted Policy}
The \textbf{mirror descent} update of $\pi_\theta$ under advantage $A(s,a)=Q(s,a)-V(s)$ is
\[
\pi'(a|s) \;\propto\; \pi_\theta(a|s)\,\exp\!\bigl(\eta\,A(s,a)\bigr).
\]
Writing $\pi_\theta(a|s)\propto e^{-E_\theta(a,s)}$, this is an \textbf{energy-based model}
\[
\pi'(a|s) \;\propto\; \exp\!\bigl(-E^{\mathrm{total}}(a,s)\bigr),
\qquad
E^{\mathrm{total}}=E_\theta - \eta\, Q.
\]

\vspace{0.3em}
\textbf{Why exactly this form?} It is the unique update satisfying
\emph{trajectory compositionality}: tilting a whole trajectory by its total
advantage equals tilting each step independently.

\vspace{0.3em}
$\Rightarrow$ Sampling from $\pi'$ \emph{correctly} (not just heuristically biasing samples)
gives a \textbf{principled policy update} -- and a tight feedback loop with the critic.
\end{block}

\vspace{0.6cm}

\begin{block}{\Large Diffusion Preliminaries}
\textbf{Forward noising:}
\[
q(x_t\mid x_0)=\mathcal{N}\!\bigl(\sqrt{\bar\alpha_t}\,x_0,\,(1-\bar\alpha_t)I\bigr).
\]
A network predicts the noise $\hat\epsilon_\theta(x_t, s, t)\approx \epsilon$, giving the
\textbf{Tweedie clean estimate}
\[
\hat x_0(x_t,t)=\frac{x_t-\sqrt{1-\bar\alpha_t}\,\hat\epsilon_\theta(x_t,s,t)}{\sqrt{\bar\alpha_t}}.
\]

\vspace{0.3em}
\textbf{Tweedie guidance} modifies the noise prediction with a clean critic:
\[
\hat\epsilon^{\mathrm{Tw}}_t = \hat\epsilon_\theta - \eta\sqrt{1-\bar\alpha_t}\,\nabla_{x_t} Q\!\bigl(s,\hat x_0(x_t,t)\bigr).
\]
Strong signal, but the guided chain is \emph{not} the reverse of any prescribed tilted
diffusion: the level-wise marginals drift away from the target. We need a corrector.
\end{block}

\end{column}

\begin{block}{\Large Energy Parameterization Unlocks MH}
We parameterize the noise prediction as the gradient of a scalar energy:
\[
\hat\epsilon_\theta(s,x_t,t)\;\coloneqq\;\sqrt{1-\bar\alpha_t}\;\nabla_{x_t} E_\theta(s,x_t,t).
\]
Training is unchanged (standard noise-MSE loss), but we now have \emph{both}
ingredients required by MCMC:
\begin{itemize}
  \item \textbf{Gradient} $\nabla_{x_t} E_\theta$ for Langevin proposals,
  \item \textbf{Energy values} $E_\theta(s,x_t,t)$ for Metropolis--Hastings ratios.
\end{itemize}

\vspace{0.3em}
The \textbf{guided level-wise energy} is
\[
\boxed{\;E^{\mathrm{guided}}_t(x) \;=\; E_\theta(s,x,t)\;-\;\eta\,Q\!\bigl(s,\hat x_0(x,t)\bigr)\;}
\]
and its gradient reproduces exactly $\hat\epsilon^{\mathrm{Tw}}_t / \sqrt{1{-}\bar\alpha_t}$.
\end{block}

\vspace{0.6cm}

\begin{block}{\Large MALA Corrector at Each Diffusion Level}
We target the level-wise density
$\;p_t(x \mid s) \propto \exp\!\bigl(-E^{\mathrm{guided}}_t(x)\bigr)\;$
with the \textbf{Metropolis-Adjusted Langevin Algorithm}:
\begin{enumerate}
  \item \textbf{Langevin propose:}
        $x' \sim \mathcal{N}\!\bigl(\mu(x),\; 2 s_t I\bigr)$,\;
        $\mu(x)=x-s_t\,\nabla E^{\mathrm{guided}}_t(x)$
  \item \textbf{MH accept/reject} with probability
        \[
        \alpha = \min\!\left\{1,\;
        e^{E^{\mathrm{guided}}_t(x)-E^{\mathrm{guided}}_t(x')}\,
        \frac{\mathcal{N}(x;\mu(x'),2s_t I)}{\mathcal{N}(x';\mu(x),2s_t I)}\right\}
        \]
  \item \textbf{Adapt} step size $s_t$ online toward the optimal $57.4\%$
        acceptance rate (Roberts et al.\ 1997).
\end{enumerate}

\vspace{0.3em}
$\Rightarrow$ MALA removes the bias of finite-step Langevin: we can take
\textbf{large} guided steps and still converge to the correct level-wise target.
\end{block}

\vspace{0.6cm}

\begin{block}{\Large \textsc{SampleFromTiltedPolicy} (MALA $+$ DDIM)}
\small
\begin{algorithmic}[1]
\REQUIRE state $s$, energy policy $E_\theta$, critic $\bar Q$, guidance $\eta$, MALA steps $M$
\STATE $\hat\epsilon_\theta(x,t)\coloneqq\sqrt{1{-}\bar\alpha_t}\,\nabla_x E_\theta(s,x,t)$
\STATE $\hat x_0(x,t)\coloneqq \mathrm{clip}\!\bigl(\tfrac{x-\sqrt{1{-}\bar\alpha_t}\,\hat\epsilon_\theta(x,t)}{\sqrt{\bar\alpha_t}},\,{\pm}r_Q\bigr)$
\STATE $E^{\mathrm{guided}}_t(x)\coloneqq E_\theta(s,x,t)-\eta\,\bar Q\!\bigl(s,\hat x_0(x,t)\bigr)$
\STATE $x_T\sim\mathcal{N}(0,I)$
\FOR{$t=T,\dots,1$}
  \STATE \textcolor{gray}{\textit{// corrector}}
  \FOR{$m=1,\dots,M$}
    \STATE $\mu \leftarrow x_t - s_t\,\nabla E^{\mathrm{guided}}_t(x_t)$; \;
           $x'\sim\mathcal{N}(\mu,2s_t I)$
    \STATE $\mu' \leftarrow x' - s_t\,\nabla E^{\mathrm{guided}}_t(x')$
    \STATE $\alpha \leftarrow \min\!\bigl\{1,\,e^{E^{\mathrm{guided}}_t(x_t)-E^{\mathrm{guided}}_t(x')}\,\tfrac{\mathcal{N}(x_t;\mu',2s_t I)}{\mathcal{N}(x';\mu,2s_t I)}\bigr\}$
    \STATE $x_t \leftarrow x'$ w.p.\ $\alpha$;\; adapt $s_t$ toward $57.4\%$
  \ENDFOR
  \STATE \textcolor{gray}{\textit{// predictor (DDIM)}}
  \STATE $\hat\epsilon_g \leftarrow \sqrt{1{-}\bar\alpha_t}\,\nabla_{x_t} E^{\mathrm{guided}}_t(x_t)$
  \STATE $x_{t-1} \leftarrow \sqrt{\tfrac{\bar\alpha_{t-1}}{\bar\alpha_t}}\,x_t + \bigl(\sqrt{1{-}\bar\alpha_{t-1}}-\sqrt{\tfrac{\bar\alpha_{t-1}(1{-}\bar\alpha_t)}{\bar\alpha_t}}\bigr)\hat\epsilon_g$
\ENDFOR
\STATE \textbf{return} $\mathrm{clip}(x_0,{-}1,1)$, updated $\{s_t\}$
\end{algorithmic}
\end{block}

\begin{block}{\Large Adaptive Guidance via a KL Budget}
Compositionality forces $\eta$ to be a \textbf{single global scalar} per update
(state-dependent $\eta(s)$ breaks trajectory tilting). For $\pi_\eta\propto \pi\, e^{\eta A}$,
\[
\KL(\pi_\eta\,\|\,\pi) \;=\; \tfrac{\eta^2}{2}\,\Var_\pi(A)\;+\;O(\eta^3).
\]

Fix a per-update KL budget $\Delta$ and invert:
\[
\boxed{\;\eta_k \;=\; \sqrt{\dfrac{2\,\Delta}{\E_{s\sim d_{\pi_k}}\!\bigl[\Var_{a\sim \pi_k}\!\bigl(A_k(s,a)\bigr)\bigr]}}\;}
\]
\begin{itemize}
  \item \textbf{Scale invariant:} $A\!\mapsto\!cA \Rightarrow \eta\!\mapsto\!\eta/c$, so $\eta A$ is unchanged.
  \item Advantage variance estimated online via an EMA of second moments.
  \item KL budget adds \emph{no} free parameter -- it just \emph{normalizes} the advantage.
\end{itemize}
\end{block}

\vspace{0.6cm}

\begin{block}{\Large Outer Loop: Online Training}
\small
\begin{algorithmic}[1]
\REQUIRE diffusion policy $E_\theta$, twin critics $Q_{\phi_1}, Q_{\phi_2}$, replay $\mathcal{D}$
\FOR{each environment step}
  \STATE $a \leftarrow \textsc{SampleFromTiltedPolicy}(s,\, E_\theta,\, \tfrac{1}{2}(Q_{\phi_1}{+}Q_{\phi_2}),\, \eta)$
  \STATE Execute $a$; push $(s,a,r,s',d)$ into $\mathcal{D}$
  \STATE Sample minibatch $\{(s_j,a_j,r_j,s'_j,d_j)\}\sim\mathcal{D}$
  \STATE $a'_j \leftarrow \textsc{SampleFromTiltedPolicy}(s'_j,\, E_\theta,\, \min_k Q_{\phi_k},\, \eta)$
  \STATE $y_j \leftarrow r_j + \gamma(1{-}d_j)\,\min_k Q_{\bar\phi_k}(s'_j, a'_j)$
  \STATE Update each $Q_{\phi_k}$ via Huber loss on $Q_{\phi_k}(s_j,a_j) - y_j$
  \STATE Distill $E_\theta$ on $(s'_j, a'_j)$ via diffusion noise-MSE
  \STATE Polyak target update; update $\eta$ from EMA of $\Var(A)$
\ENDFOR
\end{algorithmic}
\vspace{0.2em}
\textbf{Constant-weight BC.} No $Q$-reweighting in the loss --- all critic signal is
injected at inference time through MALA guidance.
\end{block}
