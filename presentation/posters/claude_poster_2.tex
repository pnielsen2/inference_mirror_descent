\begin{column}{\colwidth}

\begin{block}{\Large Mirror Descent in Policy Space}
The standard policy-improvement step in RL is the \textbf{mirror-descent} update
\[
\pi_{k+1}(a\mid s) \;\propto\; \pi_k(a\mid s)\,\exp\!\bigl(\eta\,Q^{\pi_k}(s,a)\bigr).
\]

\vspace{0.4em}
\textbf{Two ingredients needed:}
\begin{itemize}
    \item Sample from the \textbf{tilted policy} $\pi'$ at deployment time.
    \item Distill those samples back into the parametric policy $\pi_\theta$.
\end{itemize}

\vspace{0.4em}
\textbf{Why diffusion policies?}
\begin{itemize}
    \item \textbf{Expressive:} represent multimodal $\pi_\theta$ without mode collapse.
    \item \textbf{Guidable:} the reverse-diffusion chain admits gradient-based steering at inference.
    \item \textbf{Trainable by BC:} denoising score matching avoids backpropagating through the full sampling chain.
\end{itemize}

\vspace{0.4em}
\textbf{Challenge:} sampling \emph{exactly} from $\pi'\propto\pi_\theta\,e^{\eta Q}$ during the reverse process is hard --- na\"ive score addition is biased at intermediate noise levels.
\end{block}

\vspace{1cm}

\begin{block}{\Large Tilting the Reverse Diffusion}
A diffusion policy denoises actions from $\mathcal{N}(0,I)$:
\[
a_T \;\to\; a_{T-1} \;\to\; \cdots \;\to\; a_0 \sim \pi_\theta(\cdot\mid s),
\]
with the forward (noising) kernel
\[
q(a_t\mid a_0)=\mathcal{N}\!\bigl(\sqrt{\bar\alpha_t}\,a_0,\;(1-\bar\alpha_t)I\bigr).
\]

\vspace{0.3em}
\textbf{Tweedie reconstruction} maps the noisy iterate back to a clean estimate:
\[
\hat a_0(a_t)=\frac{a_t-\sqrt{1-\bar\alpha_t}\,\hat\epsilon_\theta(a_t,s,t)}{\sqrt{\bar\alpha_t}}.
\]

\vspace{0.3em}
\textbf{Training-free guidance (TFG)} evaluates a \textbf{clean} critic on $\hat a_0$:
\[
\hat\epsilon_t^{\mathrm{Tw}} = \hat\epsilon_\theta - \eta\sqrt{1-\bar\alpha_t}\,\nabla_{a_t}\,Q\!\bigl(s,\hat a_0(a_t)\bigr).
\]

\vspace{0.3em}
\textbf{Strong signal, wrong distribution:} TFG biases intermediate marginals away from the principled tilted reverse process. The endpoints are still correct, so the fix is to \emph{correct} the level-wise distribution with MCMC.
\end{block}

\vspace{1cm}

\begin{block}{\Large Key Idea: MCMC at Each Noise Level}
View the guided sampler as a sequence of \textbf{energy-based models}:
\begin{itemize}
    \item At $t=T$: target is standard normal.
    \item At $t=0$: target is the exact tilted policy $\pi'$.
    \item Intermediate $t$: smoothly interpolating EBM.
\end{itemize}

\vspace{0.3em}
\textbf{Plan:} at each level $t$, run a few MCMC steps that target the correct level-wise distribution, then take a deterministic DDIM step.
\end{block}

\end{column}

\begin{column}{\colwidth}

\begin{block}{\Large Energy-Based Parameterization}
MCMC accept/reject needs a \textbf{density ratio}, not just a score. We parameterize the policy as a scalar energy:
\[
\pi_\theta(a_t\mid s,t)\propto \exp\!\bigl(-E_\theta(s,a_t,t)\bigr),
\qquad
\hat\epsilon_\theta(s,a_t,t)=\sqrt{1-\bar\alpha_t}\,\nabla_{a_t} E_\theta(s,a_t,t).
\]
Trained with the standard denoising loss
\[
\mathcal{L}_{\mathrm{diff}}(\theta)=\mathbb{E}_{a_0,\epsilon,t}\!\Bigl[\bigl\|\hat\epsilon_\theta(s,a_t,t)-\epsilon\bigr\|^2\Bigr],
\qquad a_t=\sqrt{\bar\alpha_t}\,a_0+\sqrt{1-\bar\alpha_t}\,\epsilon.
\]

\vspace{0.3em}
\textbf{Guided level-wise energy} (the MCMC target at level $t$):
\[
E_t^{\mathrm{guided}}(a_t;\,s) \;=\; E_\theta(s,a_t,t) \;-\; \eta\, Q\!\bigl(s,\hat a_0(a_t)\bigr).
\]
Density ratios reduce to \textbf{energy differences}, so Metropolis--Hastings is tractable.
\end{block}

\vspace{1cm}

\begin{block}{\Large MALA Correction at Each Level}
At every diffusion level $t$ we run a short Metropolis-Adjusted Langevin chain targeting $p_t\propto e^{-E_t^{\mathrm{guided}}}$.

\vspace{0.3em}
\textbf{Langevin proposal:}
\[
a_t' = a_t - s_t\,\nabla_{a_t} E_t^{\mathrm{guided}}(a_t;s) + \sqrt{2 s_t}\,z,\qquad z\sim\mathcal{N}(0,I).
\]

\vspace{0.2em}
\textbf{MH accept/reject:}
\[
\alpha = \min\!\Biggl\{1,\;\exp\!\bigl(E_t^{\mathrm{guided}}(a_t)-E_t^{\mathrm{guided}}(a_t')\bigr)\,\frac{\mathcal{N}(a_t;\,a_t' - s_t\nabla E_t(a_t'),\,2s_t I)}{\mathcal{N}(a_t';\,a_t - s_t\nabla E_t(a_t),\,2s_t I)}\Biggr\}.
\]

\vspace{0.3em}
\textbf{Online step-size adaptation:} update $s_t$ after every step toward the optimal \textbf{57.4\%} acceptance rate (Roberts et al.\ 1997):
\[
\log s_t \;\leftarrow\; \log s_t + \kappa\,\bigl(\text{accept rate} - 0.574\bigr).
\]
Each diffusion level keeps its own $s_t$, learned online over training.
\end{block}

\vspace{1cm}

\begin{block}{\Large Sampling: MALA Corrector $+$ DDIM Predictor}
\algtitle{SampleFromTiltedPolicy$(s,\pi_\theta,\bar Q,\eta)$}
\vspace{-0.3em}
\begin{algorithmic}[1]
\STATE Sample $a_T\sim\mathcal{N}(0,I)$
\FOR{$t=T,T{-}1,\ldots,1$}
  \FOR{$m=1,\ldots,M$ \textbf{(MALA corrector)}}
    \STATE Propose $a_t' \sim \mathcal{N}\!\bigl(a_t - s_t\nabla E_t^{\mathrm{guided}}(a_t),\;2s_tI\bigr)$
    \STATE Accept $a_t \leftarrow a_t'$ w.p. MH ratio $\alpha$; else reject
    \STATE Adapt $s_t$ toward 57.4\% acceptance
  \ENDFOR
  \STATE DDIM step: $a_{t-1} \leftarrow \textsc{DDIM}\bigl(a_t,\hat\epsilon_t^{\mathrm{Tw}}\bigr)$
\ENDFOR
\STATE \textbf{return} $\mathrm{clip}(a_0,-1,1)$
\end{algorithmic}

\vspace{0.4em}
Endpoints are \textbf{exact}: the chain starts at $\mathcal{N}(0,I)$ and converges to $\pi'\propto\pi_\theta\,e^{\eta Q}$ as $t\to 0$. Intermediate level-wise marginals are corrected by MALA, so the sampler targets a principled mirror-descent step rather than an arbitrary high-$Q$ filter.
\end{block}

\end{column}

\begin{column}{\colwidth}

\begin{block}{\Large Online Training Loop}
At every step we (i) sample from the \emph{current} tilted policy, (ii) update twin critics by TD, and (iii) distill samples back into $\pi_\theta$ by behavior cloning.

\algtitle{MALA-Guided Mirror Descent}
\vspace{-0.5em}
\begin{algorithmic}[1]
\REQUIRE Policy $\pi_\theta$, twin critics $Q_{\phi_1},Q_{\phi_2}$, replay $\mathcal{D}$
\FOR{each env step}
  \STATE $a \leftarrow \textsc{SampleFromTiltedPolicy}\!\bigl(s,\pi_\theta,\tfrac{1}{2}(Q_{\phi_1}{+}Q_{\phi_2}),\eta\bigr)$
  \STATE Execute $a$; store $(s,a,r,s',d)$ in $\mathcal{D}$
  \STATE $a'_j \leftarrow \textsc{SampleFromTiltedPolicy}\!\bigl(s'_j,\pi_\theta,\min_k Q_{\phi_k},\eta\bigr)$
  \STATE TD target $y_j = r_j+\gamma(1{-}d_j)\min_k Q_{\bar\phi_k}(s'_j,a'_j)$
  \STATE Update $Q_{\phi_k}$: minimize $\textsc{Huber}\!\bigl(Q_{\phi_k}(s_j,a_j)-y_j\bigr)$
  \STATE BC update: fit $\pi_\theta(\cdot\mid s'_j)$ to $a'_j$ (denoising score matching)
  \STATE Polyak-average target critics
\ENDFOR
\end{algorithmic}

\vspace{0.4em}
\textbf{Twin-Q aggregation:} \emph{mean} of critics during sampling (stronger guidance signal), \emph{min} during TD bootstrapping (avoid overestimation).
\end{block}

\begin{block}{\Large Takeaways \& Future Work}
\textbf{What MGMD adds:}
\begin{itemize}
    \item Principled inference-time sampler whose limit is the mirror-descent tilted distribution.
    \item Energy parameterization unlocks MH corrections \emph{without} changing the diffusion loss.
    \item Per-level online step-size adaptation removes a major hyperparameter.
\end{itemize}

\vspace{0.1em}
\textbf{Next:} compositional $\hat Q = \mathbb{E}_{s'}[r+\gamma V]$ for multi-step planning; adaptive global $\eta$ from a KL budget; harder embodiments (Humanoid, dexterous manipulation).
\end{block}
