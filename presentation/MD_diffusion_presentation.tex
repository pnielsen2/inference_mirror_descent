\section{Algorithm}

\begin{frame}{Algorithm: MD Diffusion Policy}
\begin{fitcenter}
\begin{algorithmic}[1]
\REQUIRE Policy $\pi_\theta$ (diffusion), critics $Q_{\phi_k}$, replay buffer $\mathcal{D}$
\REQUIRE Guidance $\eta$, discount $\gamma$, target update rate $\tau_Q$
\STATE Initialize target network parameters: $\bar{\phi}_k \leftarrow \phi_k$
\FOR{each environment step}
  \STATE \textcolor{gray}{\textit{// Sample action}}
  \STATE $a \leftarrow \textsc{SampleFromTiltedPolicy}(s,\, \pi_\theta,\, \tfrac{1}{2}(Q_{\phi_1}{+}Q_{\phi_2}),\, \eta)$ \COMMENT{Alg.~2; mean Q for env}
  \STATE \textcolor{gray}{\textit{// Collect experience}}
  \STATE Execute $a$; observe reward $r$, next state $s'$, termination flag $d$
  \STATE $\mathcal{D} \leftarrow \mathcal{D} \cup \{(s, a, r, s', d)\}$
  \STATE \textcolor{gray}{\textit{// Critic update}}
  \STATE Sample minibatch $\{(s_j, a_j, r_j, s'_j, d_j)\}_{j=1}^B \sim \mathcal{D}$
  \STATE $a'_j \leftarrow \textsc{SampleFromTiltedPolicy}(s'_j,\, \pi_\theta,\, \min(Q_{\phi_1},Q_{\phi_2}),\, \eta)$ \COMMENT{min Q for TD}
  \STATE $y_j \leftarrow r_j + \gamma\,(1{-}d_j)\,\min_k(Q_{\bar{\phi}_k}(s'_j, a'_j))$
  \STATE Update each $Q_{\phi_k}$ by minimizing $\frac{1}{B}\sum_j \textsc{Huber}_{30}(Q_{\phi_k}(s_j, a_j) - y_j)$
  \STATE $\bar{\phi}_k \leftarrow \tau_Q\,\phi_k + (1{-}\tau_Q)\,\bar{\phi}_k$ for $k{=}1,2$ \COMMENT{Polyak averaging}
  \STATE \textcolor{gray}{\textit{// Policy update}}
  \STATE Fit $\pi_\theta(s'_j)$ to $a'_j$ by minimizing diffusion loss
\ENDFOR
\end{algorithmic}
\end{fitcenter}
\end{frame}

\begin{frame}{Algorithm 2: \textsc{SampleFromTiltedPolicy} (MALA $+$ DDIM)}
\begin{fitcenter}
\begin{algorithmic}[1]
\REQUIRE State $s$, energy policy $E_\theta$, aggregated critic $\bar Q(s,a)$, guidance $\eta$
\REQUIRE Log step-sizes $\{\ell_t\}$, schedule $\{\bar\alpha_t,\alpha_t,\beta_t\}$, clips $r$ (Q-eval), $r_0$ (recon), adapt rate $\rho$
\STATE $\hat\epsilon_\theta(x,t)\coloneqq\nabla_x E_\theta(s,x,t)$;\quad $\sigma_t\coloneqq\sqrt{1{-}\bar\alpha_t}$;\quad $\hat x_0(x,t)\coloneqq(x-\sigma_t\hat\epsilon_\theta(x,t))/\sqrt{\bar\alpha_t}$ \COMMENT{Tweedie}
\STATE $U_t(x)\coloneqq E_\theta(s,x,t)-\eta\cdot\bar Q(s,\,\mathrm{clip}(\hat x_0(x,t),{-}r,r))$ \COMMENT{tilted energy}
\STATE Sample $x_T\sim\mathcal{N}(0,I)$
\FOR{$t=T{-}1,\dots,0$}
  \FOR{$m=1,\dots,M$}
    \STATE \textcolor{gray}{\textit{// MALA corrector ($M{=}2$ steps at this noise level)}}
    \STATE $\eta_t\leftarrow\mathrm{clip}(\exp(\ell_t)\beta_t,\;10^{-8},\;0.5)$
    \STATE $\mu\leftarrow x_t-\eta_t\nabla U_t(x_t)$;\quad $x'\leftarrow\mu+\sqrt{2\eta_t}\,\xi$,\;\;$\xi\sim\mathcal{N}(0,I)$
    \STATE $\mu'\leftarrow x'-\eta_t\nabla U_t(x')$;\quad $\log\alpha\leftarrow(U_t(x_t){-}U_t(x'))+\tfrac{\|x'{-}\mu\|^2-\|x_t{-}\mu'\|^2}{4\eta_t}$ \COMMENT{MALA MH}
    \STATE $x_t\leftarrow x'$ if $\log u<\min(0,\log\alpha)$,\;\;$u\sim\mathrm{Uniform}(0,1)$
    \STATE $\ell_t\leftarrow\mathrm{clip}(\ell_t{+}\rho(\mathbf{1}_{\mathrm{acc}}{-}0.574),\;\ell_{\min},\;\ell_{\max})$
  \ENDFOR
  \STATE \textcolor{gray}{\textit{// Guided DDIM predictor:}}
  \STATE $g\leftarrow\nabla_{x_t}\bar Q(s,\,\mathrm{clip}(\hat x_0(x_t,t),{-}r,r))$;\quad $\hat\epsilon_g\leftarrow\hat\epsilon_\theta(x_t,t)-\eta\sigma_t g$
  \STATE $\hat x_0^g\leftarrow\mathrm{clip}\!\left(\tfrac{x_t-\sigma_t\hat\epsilon_g}{\sqrt{\bar\alpha_t}},\;{-}r_0,\;r_0\right)$;\quad $x_{t-1}\leftarrow\tfrac{\beta_t\sqrt{\bar\alpha_{t-1}}}{1{-}\bar\alpha_t}\hat x_0^g+\tfrac{(1{-}\bar\alpha_{t-1})\sqrt{\alpha_t}}{1{-}\bar\alpha_t}x_t$
\ENDFOR
\STATE \textbf{return} $\mathrm{clip}(x_0,{-}1,1)$, updated $\{\ell_t\}$
\end{algorithmic}
\end{fitcenter}
\end{frame}

\begin{frame}{Algorithm 2: Mathematical Foundations}
\small
\textbf{What MALA targets at each level.}\; MALA samples from the $Q$-tilted EBM $p_t(x\mid s)\propto\exp(-U_t(x))$, where $U_t$ is defined in Alg.~2. The EBM sequence interpolates from a Gaussian ($t{=}T$) to the exact tilted distribution $\pi'\propto\pi_\theta e^{\eta Q}$ at $t{=}0$.

\smallskip
\textbf{Why energy-based?}\; MALA requires the density ratio $p_t(x')/p_t(x)$; score networks give only $\nabla\log p_t$. The energy MLP $E_\theta$ outputs a scalar, so the ratio is computable:
\[
  \frac{p_t(x')}{p_t(x)} = \exp\!\bigl(U_t(x)-U_t(x')\bigr)
\]
This is the first term of $\log\alpha$ in Alg.~2. The second term $\smash{\tfrac{\|x'-\mu\|^2-\|x_t-\mu'\|^2}{4\eta_t}}$ corrects for the asymmetric gradient-biased proposal; pure Langevin dynamics (without this correction) has $O(\eta_t)$ bias.

\smallskip
\textbf{Why Tweedie for $\bar Q$?}\; At $t{=}0$, na\"{i}ve score summation is exact: $\nabla\log(p_0\,e^{\eta Q})=\nabla\log p_0+\eta\nabla Q$. At $t{>}0$, the correct tilted score requires the intractable marginal $\tilde p_t(x_t)=\int p_{t|0}(x_t|x_0)\,\tilde p_0(x_0)\,dx_0$, so $\nabla\log\tilde p_t\neq\nabla\log p_t+\eta\nabla Q(s,\hat x_0)$. The Tweedie estimate $\hat x_0(x_t,t)=\mathbb{E}[x_0|x_t]$ is a tractable substitute; MALA's accept/reject absorbs the approximation error.

\end{frame}
